def train_isolation_forest(df):
    X = df[FEATURES].fillna(0)
    model = IsolationForest(
        n_estimators=250,
        contamination=0.03,
        random_state=42
    )
    out = df.copy()
    out["Anomaly_Prediction"] = model.fit_predict(X)
    out["Anomaly_Score"] = -model.score_samples(X)
    out["Anomaly"] = (out["Anomaly_Prediction"] == -1).astype(int)
    return out, model

def run_lof(df):
    X = df[FEATURES].fillna(0)
    lof = LocalOutlierFactor(n_neighbors=25, contamination=0.03)
    out = df.copy()
    out["LOF_Anomaly"] = (lof.fit_predict(X) == -1).astype(int)
    return out

from sklearn.cluster import KMeans, DBSCAN
from sklearn.preprocessing import StandardScaler
import joblib

FEATURES = ["Latitude", "Longitude", "Hour", "Speed_kmh", "Distance_km"]

def train_kmeans(df, n_clusters=8):
    X = df[FEATURES].fillna(0)
    scaler = StandardScaler()
    Xs = scaler.fit_transform(X)

    model = KMeans(n_clusters=n_clusters, random_state=42, n_init=10)
    df = df.copy()
    df["Cluster"] = model.fit_predict(Xs)

    return df, model, scaler

def train_dbscan(df):
    X = df[["Latitude", "Longitude"]].fillna(0)
    scaler = StandardScaler()
    Xs = scaler.fit_transform(X)
    model = DBSCAN(eps=0.35, min_samples=8)
    labels = model.fit_predict(Xs)
    out = df.copy()
    out["DBSCAN_Cluster"] = labels
    return out, model, scaler

# Fictional study region: Bhopal-like coordinate range.
AREAS = {
    "Area_A": (23.2599, 77.4126),
    "Area_B": (23.2350, 77.4340),
    "Area_C": (23.2850, 77.3500),
    "Area_D": (23.2150, 77.3900),

}

def generate_movement(n_users=35, days=35):
    rows = []
    area_names = list(AREAS)

    for uid in range(1, n_users + 1):
        preferred = rng.choice(area_names)
        for day in range(days):
            n_points = int(rng.integers(18, 34))
            start = pd.Timestamp("2026-07-01") + pd.Timedelta(days=day)
            times = [start + pd.Timedelta(minutes=int(x)) for x in sorted(
                rng.choice(np.arange(0, 24 * 60), size=n_points, replace=False)
            )]

            current_area = preferred
            for ts in times:
                if rng.random() < 0.18:
                    current_area = rng.choice(area_names)

                lat0, lon0 = AREAS[current_area]
                lat = lat0 + rng.normal(0, 0.0045)
                lon = lon0 + rng.normal(0, 0.0045)

                speed = max(0.5, rng.normal(22 if 7 <= ts.hour <= 20 else 12, 5))
                rows.append([
                    f"U{uid:03d}", ts, lat, lon, speed,
                    "Friday" if ts.dayofweek == 4 else ts.day_name(),
                    "Rain" if rng.random() < 0.18 else "Clear",
                    current_area
                ])

    df = pd.DataFrame(rows, columns=[
        "User_ID", "Timestamp", "Latitude", "Longitude",
        "Speed_kmh", "Day", "Weather", "Area"
    ])
    df["Timestamp"] = pd.to_datetime(df["Timestamp"])
    df = df.sort_values(["User_ID", "Timestamp"]).reset_index(drop=True)

    # Consecutive distance using a small-region approximation.
    dlat = df.groupby("User_ID")["Latitude"].diff()
    dlon = df.groupby("User_ID")["Longitude"].diff()
    df["Distance_km"] = np.sqrt((dlat * 111)**2 + (dlon * 102)**2).fillna(0)
    df["Hour"] = df["Timestamp"].dt.hour
    df["Weekday"] = df["Timestamp"].dt.dayofweek
    df["Weekend"] = (df["Weekday"] >= 5).astype(int)

    # Add a small number of naturally unusual points for anomaly-model demonstration.
    idx = rng.choice(df.index, size=max(10, len(df)//250), replace=False)
    df.loc[idx, "Speed_kmh"] *= rng.uniform(2.5, 4.0, size=len(idx))
    df.loc[idx, "Distance_km"] *= rng.uniform(2.0, 5.0, size=len(idx))

    return df

def generate_cases(movement):
    stats = movement.groupby("User_ID").agg(
        Average_Distance=("Distance_km", "mean"),
        Average_Speed=("Speed_kmh", "mean"),
        Last_Latitude=("Latitude", "last"),
        Last_Longitude=("Longitude", "last"),
        Usual_Area=("Area", lambda s: s.mode().iloc[0]),
        Previous_Area=("Area", lambda s: s.iloc[-2] if len(s) > 1 else s.iloc[-1]),
        Last_Seen_Time=("Timestamp", "last")
    ).reset_index()

    cases = []
    for i, row in stats.iterrows():
        cases.append({
            "Case_ID": f"MP-2026-{17+i:03d}",
            "Person_ID": f"P-{1001+i}",
            "Age_Group": rng.choice(["18-25", "26-35", "36-45"]),
            "Gender": rng.choice(["Female", "Male"]),
            "Last_Latitude": row.Last_Latitude,
            "Last_Longitude": row.Last_Longitude,
            "Last_Seen_Time": row.Last_Seen_Time,
            "Day": row.Last_Seen_Time.day_name(),
            "Weather": "Rain" if row.Last_Seen_Time.dayofweek % 3 == 0 else "Clear",
            "Usual_Area": row.Usual_Area,
            "Average_Distance": round(row.Average_Distance, 3),
            "Average_Speed": round(row.Average_Speed, 3),
            "Previous_Area": row.Previous_Area,
            "Time_Since_Last_Seen": int(rng.integers(1, 9)),
            "Target_Area": rng.choice(list(AREAS.keys()))
        })
    return pd.DataFrame(cases)

import pandas as pd
import numpy as np
import joblib
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, confusion_matrix

FEATURES = [
    "Age_Group", "Day", "Hour", "Weather", "Usual_Area",
    "Previous_Area", "Last_Latitude", "Last_Longitude",
    "Average_Distance", "Average_Speed", "Time_Since_Last_Seen"
]
TARGET = "Target_Area"

def train_location_model(cases):
    data = cases.copy()
    data["Hour"] = pd.to_datetime(data["Last_Seen_Time"]).dt.hour

    # Synthetic labels are deliberately generated for academic demonstration.
    # In a real study, target labels would require a properly documented dataset.
    X = data[FEATURES]
    y = data[TARGET]

    categorical = ["Age_Group", "Day", "Weather", "Usual_Area", "Previous_Area"]
    numeric = [c for c in FEATURES if c not in categorical]

    pre = ColumnTransformer([
        ("cat", OneHotEncoder(handle_unknown="ignore"), categorical),
        ("num", StandardScaler(), numeric)
    ])

    model = RandomForestClassifier(
        n_estimators=300,
        max_depth=12,
        min_samples_leaf=2,
        class_weight="balanced",
        random_state=42
    )

    Xtr, Xte, ytr, yte = train_test_split(
        X, y, test_size=0.25, random_state=42, stratify=y
    )

    Xtr2 = pre.fit_transform(Xtr)
    Xte2 = pre.transform(Xte)
    model.fit(Xtr2, ytr)

    pred = model.predict(Xte2)
    metrics = {
        "accuracy": float(accuracy_score(yte, pred)),
        "precision_weighted": float(precision_score(yte, pred, average="weighted", zero_division=0)),
        "recall_weighted": float(recall_score(yte, pred, average="weighted", zero_division=0)),
        "f1_weighted": float(f1_score(yte, pred, average="weighted", zero_division=0)),
        "confusion_matrix": confusion_matrix(yte, pred).tolist()
    }

    return data, pre, model, metrics

def predict_case(case_row, pre, model):
    row = pd.DataFrame([case_row]).copy()
    row["Hour"] = pd.to_datetime(row["Last_Seen_Time"]).dt.hour
    X = pre.transform(row[FEATURES])
    probs = model.predict_proba(X)[0]
    classes = model.classes_
    result = pd.DataFrame({"Area": classes, "Probability": probs})
    return result.sort_values("Probability", ascending=False).reset_index(drop=True)

def make_map(last_lat, last_lon, frequent_areas, predicted_df, route=None):
    m = folium.Map(location=[last_lat, last_lon], zoom_start=12)

    folium.Marker(
        [last_lat, last_lon],
        tooltip="Last known simulated location",
        icon=folium.Icon(icon="info-sign")
    ).add_to(m)

    for area in frequent_areas:
        if area in AREA_COORDS:
            lat, lon = AREA_COORDS[area]
            folium.CircleMarker(
                [lat, lon], radius=7,
                tooltip=f"Frequent area: {area}"
            ).add_to(m)

    for _, row in predicted_df.iterrows():
        area = row["Area"]
        if area in AREA_COORDS:
            lat, lon = AREA_COORDS[area]
            folium.Marker(
                [lat, lon],
                tooltip=f"{area} | Probability {row['Probability']:.1%}"
            ).add_to(m)

    if route:
        coords = [AREA_COORDS[a] for a in route if a in AREA_COORDS]
        if len(coords) >= 2:
            folium.PolyLine(coords, tooltip="Simulated Markov Chain route").add_to(m)

    return m


import pandas as pd
import numpy as np

def clean_movement(df):
    df = df.copy()
    df["Timestamp"] = pd.to_datetime(df["Timestamp"], errors="coerce")
    df = df.dropna(subset=["Timestamp", "Latitude", "Longitude"])
    df = df.drop_duplicates()

    df = df[
        df["Latitude"].between(-90, 90) &
        df["Longitude"].between(-180, 180)
    ].copy()

    for col in ["Speed_kmh", "Distance_km"]:
        if col in df:
            df[col] = pd.to_numeric(df[col], errors="coerce").fillna(df[col].median())

    df["Hour"] = df["Timestamp"].dt.hour
    df["Weekday"] = df["Timestamp"].dt.dayofweek
    df["Month"] = df["Timestamp"].dt.month
    df["Weekend"] = (df["Weekday"] >= 5).astype(int)
    df["Day_Name"] = df["Timestamp"].dt.day_name()

    return df.sort_values(["User_ID", "Timestamp"]).reset_index(drop=True)

def add_movement_features(df):
    df = df.copy()
    g = df.groupby("User_ID", group_keys=False)

    df["Prev_Latitude"] = g["Latitude"].shift(1)
    df["Prev_Longitude"] = g["Longitude"].shift(1)

    dlat = df["Latitude"] - df["Prev_Latitude"]
    dlon = df["Longitude"] - df["Prev_Longitude"]
    df["Consecutive_Distance_km"] = np.sqrt((dlat * 111)**2 + (dlon * 102)**2).fillna(0)

    df["Movement_Frequency"] = g["Area"].transform("count")
    df["Area_Visit_Frequency"] = df.groupby(["User_ID", "Area"])["Area"].transform("count")
    df["Total_User_Distance"] = g["Distance_km"].transform("sum")
    df["Avg_User_Speed"] = g["Speed_kmh"].transform("mean")
    df["Max_User_Speed"] = g["Speed_kmh"].transform("max")

    return df


import pandas as pd
import numpy as np

def minmax(series):
    s = pd.Series(series, dtype=float)
    if s.max() == s.min():
        return pd.Series(np.ones(len(s)), index=s.index)
    return (s - s.min()) / (s.max() - s.min())

def calculate_priority(prediction, history_frequency, route_similarity,
                       distance_relevance, time_relevance, anomaly_evidence):
    # Guideline weights:
    # prediction 30%, history 20%, route 15%, distance 15%, time 10%, anomaly 10%.
    score = (
        0.30 * prediction +
        0.20 * history_frequency +
        0.15 * route_similarity +
        0.15 * distance_relevance +
        0.10 * time_relevance +
        0.10 * anomaly_evidence
    ) * 100

    band = np.select(
        [score <= 30, score <= 60, score <= 80],
        ["Low", "Medium", "High"],
        default="Very High"
    )
    return score, band

def rank_locations(probabilities, historical_counts, route_scores=None,
                    distance_scores=None, time_scores=None, anomaly_scores=None):
    areas = probabilities.index.astype(str)
    n = len(areas)
    route_scores = route_scores if route_scores is not None else np.zeros(n)
    distance_scores = distance_scores if distance_scores is not None else np.zeros(n)
    time_scores = time_scores if time_scores is not None else np.zeros(n)
    anomaly_scores = anomaly_scores if anomaly_scores is not None else np.zeros(n)

    hist = minmax(historical_counts.values)
    pred = minmax(probabilities.values)

    score, band = calculate_priority(
        pred, hist, route_scores, distance_scores, time_scores, anomaly_scores
    )
    out = pd.DataFrame({
        "Area": areas,
        "Prediction_Probability": probabilities.values,
        "Historical_Frequency": historical_counts.values,
        "Search_Priority": score,
        "Priority_Band": band
    })
    return out.sort_values("Search_Priority", ascending=False).reset_index(drop=True)

import pandas as pd
import numpy as np
import joblib
from collections import defaultdict

class MarkovRouteModel:
    """First-order Markov Chain: P(next_area | current_area)."""

    def __init__(self):
        self.transition_counts = {}
        self.probabilities = {}

    def fit(self, df):
        ordered = df.sort_values(["User_ID", "Timestamp"])
        for _, user_df in ordered.groupby("User_ID"):
            areas = user_df["Area"].dropna().tolist()
            for a, b in zip(areas[:-1], areas[1:]):
                if a != b:
                    self.transition_counts.setdefault(a, {})
                    self.transition_counts[a][b] = self.transition_counts[a].get(b, 0) + 1

        for a, counts in self.transition_counts.items():
            total = sum(counts.values())
            self.probabilities[a] = {
                b: n / total for b, n in counts.items()
            }
        return self

    def next_locations(self, current_area, top_n=5):
        vals = self.probabilities.get(current_area, {})
        return sorted(vals.items(), key=lambda x: x[1], reverse=True)[:top_n]

    def predict_route(self, start_area, steps=4):
        route = [start_area]
        probs = []
        current = start_area
        for _ in range(steps):
            candidates = self.next_locations(current, 1)
            if not candidates:
                break
            nxt, p = candidates[0]
            route.append(nxt)
            probs.append(p)
            current = nxt
        return route, probs

def save_model(model, path):
    joblib.dump(model, path)
